2Names & values

See Lesson 02 — Names and values

Assignment binds a name to a value
score = 10          # name = value
score = score + 5   # rebind to a new value (15)

a = 1
b = a               # b points at the same value

3Core types

See Lesson 03 — The core types

int, float, bool, None
type(42)        # <class 'int'>
type(3.14)      # <class 'float'>
type(True)      # <class 'bool'>
type(None)      # <class 'NoneType'>

int("7")        # 7        (string -> int)
float("2.5")    # 2.5
str(42)         # "42"
bool(0)         # False    (0, "", [], None are falsy)

7 / 2           # 3.5   true division
7 // 2          # 3     floor division
7 % 2           # 1     remainder
2 ** 10         # 1024  power

4Strings & f-strings

See Lesson 04 — Strings and f-strings

Immutable text
name = "Ada"
name[0]            # "A"      index from 0
name[-1]           # "a"      last character
name[0:2]          # "Ad"     slice (end excluded)
len(name)          # 3

name.upper()       # "ADA"
name.lower()       # "ada"
"  hi  ".strip()   # "hi"
"a,b,c".split(",") # ["a", "b", "c"]

f"{name} is {len(name)}"   # "Ada is 3"

# strings are immutable: name[0] = "B" raises TypeError

5Lists & indexing

See Lesson 05 — Lists and indexing

Ordered, mutable sequences
nums = [10, 20, 30]
nums[0]            # 10
nums[-1]           # 30
nums[0:2]          # [10, 20]
len(nums)          # 3

nums.append(40)    # [10, 20, 30, 40]
nums[0] = 99       # lists ARE mutable
20 in nums         # True

for n in nums:     # iterate values
    print(n)

for i, n in enumerate(nums):   # index + value
    print(i, n)

7Loops & repetition

See Lesson 07 — Loops and repetition

for and while
for i in range(3):     # 0, 1, 2
    print(i)

range(2, 6)            # 2, 3, 4, 5
range(0, 10, 2)        # 0, 2, 4, 6, 8

n = 3
while n > 0:           # repeat while true
    print(n)
    n -= 1

for n in nums:
    if n == 20:
        break          # stop the loop
    if n == 10:
        continue       # skip to next

8Defining functions

See Lesson 08 — Defining functions

def, parameters, return
def double(n):
    """Return n times two."""
    return n * 2

double(5)          # 10

def area(w, h):
    return w * h

area(3, 4)         # 12   positional
area(h=4, w=3)     # 12   keyword

# no return -> returns None
def log(msg):
    print(msg)

9Scope & arguments

See Lesson 09 — Scope and arguments

LEGB & defaults
# lookup order: Local, Enclosing, Global, Built-in
def f():
    x = 1          # local — gone after the call
    return x

def greet(name, greeting="Hello"):   # default
    return f"{greeting}, {name}!"

greet("Ada")                 # "Hello, Ada!"
greet("Ada", greeting="Hi")  # "Hi, Ada!"

# TRAP: never default to a mutable value
def add(item, items=None):   # GOOD
    if items is None:
        items = []
    items.append(item)
    return items

10Modules & the standard library

See Lesson 10 — Modules and the standard library

Import styles & __main__
import math                    # math.sqrt(16)
from math import sqrt          # sqrt(16)
from math import sqrt as root  # root(16)

# standard library (no install needed):
#   math, random, datetime, json, os

def main():
    print("running")

if __name__ == "__main__":     # only when run directly
    main()

# avoid: from math import *   (dumps every name)
