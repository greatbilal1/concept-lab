1G-Counter CRDT Implementation

Conflict-free distributed counter merge
class GCounter:
    def __init__(self, node_id, n=3):
        self.node_id = node_id
        self.counts = [0] * n
    def inc(self): self.counts[self.node_id] += 1
    def value(self): return sum(self.counts)
    def merge(self, other):
        # Commutative, associative, idempotent maximum:
        self.counts = [max(a, b) for a, b in zip(self.counts, other.counts)]

2Consistent Hashing Ring Lookup

Clockwise binary search on ring
import bisect, hashlib
def get_node(key, ring_keys, ring_map):
    h = int(hashlib.sha256(key.encode()).hexdigest(), 16)
    idx = bisect.bisect_right(ring_keys, h)
    if idx == len(ring_keys): idx = 0 # Wrap around ring!
    return ring_map[ring_keys[idx]]

3Leaderless Quorum Assertion

Dynamo-style strong consistency condition
# To guarantee linearizable reads in Dynamo/Cassandra:
N = 3 # Total replica nodes
W = 2 # Write acknowledged by 2 nodes

assert (W + R) > N, 'Quorum condition violated! Stale reads possible!'
