1Locating Entry Points

Find where execution starts
# Python CLI / Web
grep -rn "if __name__ == '__main__':" .
grep -rn "FastAPI(" . || grep -rn "Flask(" .

# Node.js
jq .main package.json
jq .scripts.start package.json
