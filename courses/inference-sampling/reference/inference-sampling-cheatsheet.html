3Stable Softmax with Temperature

NumPy implementation
import numpy as np
def stable_softmax(logits, temperature=1.0):
    scaled = logits / max(temperature, 1e-5)
    exp_s = np.exp(scaled - np.max(scaled))
    return exp_s / np.sum(exp_s)

4Top-P (Nucleus) Truncation Algorithm

Cumulative mass filtering
def top_p_filter(probs, p=0.9):
    sorted_indices = np.argsort(probs)[::-1]
    sorted_probs = probs[sorted_indices]
    cum_probs = np.cumsum(sorted_probs)
    # Keep tokens where previous cumulative sum is < p:
    cutoff = cum_probs > p
    cutoff[1:] = cutoff[:-1]
    cutoff[0] = False
    sorted_probs[cutoff] = 0.0
    return sorted_probs / np.sum(sorted_probs)
