2Secret Token Leak Regex Scanner

Egress credential scanner
import re
SECRET_PATTERN = r'(sk-[a-zA-Z0-9]{32,}|AKIA[0-9A-Z]{16}|ghp_[a-zA-Z0-9]{36})'
def scan_secrets(output_text):
    if re.search(SECRET_PATTERN, output_text):
        raise SecurityException('Credential leak detected in output!')

3Guardrails AI Policy Composition

Modular Python validation

from guardrails.hub import ProfanityFree, ToxicLanguage
guard = Guard().use_many(
    ProfanityFree(on_fail='filter'),
    ToxicLanguage(threshold=0.8, on_fail='fix')
)

4Strict Domain Egress Allow-List

Preventing SSRF attacks in agent scraping
ALLOWED_HOSTS = {'docs.stripe.com', 'api.github.com'}
def safe_fetch(target_url):
    domain = urllib.parse.urlparse(target_url).netloc
    if domain not in ALLOWED_HOSTS:
        raise SecurityException(f'Access to {domain} forbidden!')
    return requests.get(target_url)
