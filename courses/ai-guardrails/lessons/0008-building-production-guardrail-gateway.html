async def secure_chat_gateway(request: ChatRequest, user: User = Depends(auth)):
    # 1. Input Guardrail: PII + Safety + Topicality
    scrubbed_prompt = scrub_pii(request.prompt)
    if not await check_safety_and_topic(scrubbed_prompt):
        return SafeRejectionResponse("Request violates safety or scope policies.")
        
    # 2. Model Execution via Resilient Proxy
    raw_response = await model_router.dispatch(request.model, scrubbed_prompt)
    
    # 3. Output Guardrail: Secrets + Schema + Grounding
    if contains_secrets(raw_response.text):
        alert_soc_and_escalate(user.id, "Credential Leak Attempt")
        return SafeRejectionResponse("Response blocked by security policy.")
        
    validated_data = validate_schema(raw_response.text, request.response_format)
    return validated_data
