1OpenAI Pydantic Structured Output Call

100% guaranteed schema contract
from pydantic import BaseModel, Field

class OrderData(BaseModel):
    order_id: str = Field(description="Order ID like ORD-12345")
    total_cents: int = Field(ge=0, description="Price in integer cents")

# Guaranteed Pydantic instance:
res = client.beta.chat.completions.parse(
    model="gpt-4o",
    messages=[{"role": "user", "content": prompt}],
    response_format=OrderData
)
data: OrderData = res.choices[0].message.parsed

2Self-Healing JSON Repair Loop

Automated error recovery pattern
from pydantic import ValidationError
for attempt in range(3):
    res = call_llm(prompt)
    try:
        return MySchema.model_validate_json(res)
    except ValidationError as err:
        prompt = f"Validation failed:\n{err.json()}\nPlease fix and return valid JSON."

3Hierarchical Nested Schema Pattern

Composing child models
class LineItem(BaseModel):
    sku: str
    qty: int = Field(gt=0)

class Invoice(BaseModel):
    invoice_id: str
    items: list[LineItem] = Field(min_length=1)

4Defensive Nullable Attribute

Preventing forced hallucinations
class Customer(BaseModel):
    name: str
    # Nullable with default prevents model from hallucinating fake phone:
    phone: str | None = Field(default=None, description="Phone if stated, else null")
