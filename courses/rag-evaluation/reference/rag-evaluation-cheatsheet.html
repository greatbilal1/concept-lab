1Automated Ragas Evaluation Run

Computing RAG Triad metrics
from ragas import evaluate
from ragas.metrics import faithfulness, answer_relevancy, context_recall
from datasets import Dataset

dataset = Dataset.from_dict(eval_records)
results = evaluate(dataset, metrics=[faithfulness, answer_relevancy, context_recall])
df = results.to_pandas()
print(f"Faithfulness: {df['faithfulness'].mean():.2f}")

3Synthetic Testset Generation Script

Bootstrapping 50 test cases in 2 minutes
from ragas.testset.generator import TestsetGenerator
generator = TestsetGenerator.with_openai()
testset = generator.generate_with_langchain_docs(documents, test_size=50)
testset.to_pandas().to_json('evals/golden.jsonl', orient='records')
