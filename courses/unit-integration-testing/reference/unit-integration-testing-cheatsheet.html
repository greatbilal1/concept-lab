1Sociable In-Memory Unit Test

Testing real classes in RAM
def test_order_total():
    cart = Cart([Item(price=50, qty=2)])
    calc = PriceCalculator(tax_rate=0.1)
    assert calc.total(cart) == 110.0

2HTTP Mocking with Responses

Intercepting network requests
@responses.activate
def test_api_call():
    responses.add(responses.GET, "https://api.com/u/1", json={"id": 1}, status=200)
    res = client.get_user(1)
    assert res.id == 1

4FastAPI TestClient Request

In-process ASGI endpoint test
from fastapi.testclient import TestClient
client = TestClient(app)

def test_health():
    res = client.get("/health")
    assert res.status_code == 200
    assert res.json() == {"status": "ok"}
