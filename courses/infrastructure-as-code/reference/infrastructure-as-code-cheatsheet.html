1Core CLI Workflow

Lifecycle commands
# Initialize providers and backend
terraform init

# Preview changes against current state
terraform plan -out=tfplan

# Apply verified execution plan
terraform apply tfplan

# Inspect current managed state
terraform state list
terraform show

See Lesson 02 — Terraform Architecture & Workflow.

2Remote State & Locking

Backend configuration
terraform {
  required_version = ">= 1.5.0"
  backend "s3" {
    bucket         = "corp-tf-state-prod"
    key            = "vpc/terraform.tfstate"
    region         = "us-east-1"
    dynamodb_table = "terraform-locks" # State locking
    encrypt        = true
  }
}

See Lesson 03 — State Management & Backends.

3Resource & Variables

Declarative resource block
variable "environment" {
  type        = string
  description = "Target deployment environment"
  default     = "production"
}

resource "aws_s3_bucket" "data_lake" {
  bucket = "company-data-${var.environment}"
  tags = {
    Environment = var.environment
    ManagedBy   = "Terraform"
  }
}

output "bucket_arn" {
  value       = aws_s3_bucket.data_lake.arn
  description = "The Amazon Resource Name of the bucket"
}

See Lesson 04 — Variables, Outputs & Locals.

4Module Instantiation

Reusable infrastructure block
module "vpc" {
  source = "./modules/vpc"

  cidr_block   = "10.0.0.0/16"
  subnet_count = 3
  environment  = "prod"
}

output "vpc_id" {
  value = module.vpc.id
}
