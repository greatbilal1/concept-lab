2Client Disconnect Detection in Streaming

Cancelling ghost streams to save tokens
async def sse_generator(request: Request, prompt: str):
    async for chunk in llm.stream(prompt):
        if await request.is_disconnected():
            logger.info('Client closed tab. Aborting generation!')
            break
        yield f"data: {json.dumps({'t': chunk.text})}\n\n"

def search_tenant_docs(query_vector, tenant_id):
    return vector_index.query(

4Redis Sliding Window Session Push

Stateless chat memory maintenance
async def save_turn(session_id, user_msg, ai_msg, max_turns=10):
    key = f"session:{session_id}:window"
    await redis.rpush(key, json.dumps({'u': user_msg, 'a': ai_msg}))
    await redis.ltrim(key, -max_turns, -1) # Keep last N turns
    await redis.expire(key, 86400 * 7) # 7-day TTL

See Lesson 04 — Distributed Context and Session Storage (Redis, PostgreSQL).
