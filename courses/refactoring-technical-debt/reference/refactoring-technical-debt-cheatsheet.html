2Immutable Value Object Pattern

Curing primitive obsession
from dataclasses import dataclass

@dataclass(frozen=True)
class Money:
    amount: float
    currency: str = 'USD'
    def __post_init__(self):
        if self.amount < 0: raise ValueError('Negative money forbidden')
