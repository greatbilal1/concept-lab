1FastAPI Server-Sent Events Streaming

Low-latency token streaming
from fastapi.responses import StreamingResponse
@app.post("/chat/stream")
async def stream(prompt: str):
    async def gen():
        stream = await client.chat.completions.create(model="gpt-4o-mini", messages=[...], stream=True)
        async for chunk in stream:
            yield f"data: {json.dumps({'token': chunk.choices[0].delta.content or ''})}\n\n"
    return StreamingResponse(gen(), media_type="text/event-stream")

2Two-Tier Model Cascade Pattern

Cost-optimized routing fallback
# 1. Try fast cheap tier ($0.15/1M):
res = await call_fast_model(prompt, max_tokens=150)
if res.confidence >= 0.90 and not res.expressed_doubt:
    return res.text # 80% resolved here!
# 2. Escalate remaining 20% to frontier tier ($5.00/1M):
return await call_frontier_model(prompt, max_tokens=600)

3Semantic Vector Cache Lookup

15ms sub-cent response reuse
vector = embed(query)
match = await redis_vector.find_nearest(vector, threshold=0.96)
if match:
    return match.cached_response # 15ms cache hit!
answer = await call_llm(query)
await redis_vector.save(vector, answer, ttl=86400)
return answer
