1Streaming Instantiation

Fastest way to compile and load Wasm
// Stream and compile simultaneously while downloading
const { instance, module } = await WebAssembly.instantiateStreaming(
  fetch("math.wasm"),
  {
    env: {
      log: (val) => console.log("From Wasm:", val)
    }
  }
);

// Call exported Wasm function
const sum = instance.exports.add(40, 2);

2Handwritten WAT Module

WebAssembly text syntax
(module
  ;; Import JS function
  (import "env" "log" (func $log (param i32)))
  
  ;; Export add function
  (func (export "add") (param $a i32) (param $b i32) (result i32)
    local.get $a
    local.get $b
    i32.add
  )
)

3Linear Memory Access

Reading bytes across the boundary
// JavaScript allocating 64KB Wasm page
const memory = new WebAssembly.Memory({ initial: 1 }); // 1 page = 64KB
const view = new Uint8Array(memory.buffer);

// Write text into Wasm memory
const encoder = new TextEncoder();
view.set(encoder.encode("Hello Wasm!"), 0);

// Pass pointer (offset 0) and length into Wasm
instance.exports.process_string(0, 11);

4Rust wasm-bindgen Export

Idiomatic compiled Rust
use wasm_bindgen::prelude::*;

#[wasm_bindgen]
pub fn fibonacci(n: u32) -> u32 {
    match n {
        0 => 0,
        1 => 1,
        _ => fibonacci(n - 1) + fibonacci(n - 2),
    }
}
