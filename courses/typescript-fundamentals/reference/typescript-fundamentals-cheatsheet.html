2any vs unknown vs never

Safe top and bottom types
// any disables the compiler completely (avoid!)
let bad: any = "hello";
bad.nonExistentMethod(); // compiles, crashes at runtime!

// unknown requires narrowing before use (safe!)
let input: unknown = JSON.parse('{"id": 1}');
if (typeof input === "object" && input !== null && "id" in input) {
  console.log((input as { id: number }).id);
}

// never represents values that can never occur
function fail(msg: string): never {
  throw new Error(msg);
}

See Lesson 02 — Type Inference & Safe Types.

3Interfaces vs Type Aliases

Modeling object shapes
interface User {
  readonly id: string;
  name: string;
  email?: string; // optional property
}

// Interfaces support declaration merging and extension
interface Admin extends User {
  role: "superadmin" | "moderator";
}

// Type aliases support unions, primitives, and mapped types
type ID = string | number;
type Callback = (data: string) => void;

See Lesson 03 — Interfaces & Object Shapes.

4Discriminated Unions

Exhaustive state modeling
type NetworkState =
  | { state: "idle" }
  | { state: "loading" }
  | { state: "success"; data: string }
  | { state: "error"; message: string };

function render(s: NetworkState) {
  switch (s.state) {
    case "idle": return "Ready";
    case "loading": return "Loading...";
    case "success": return s.data; // TS knows data exists!
    case "error": return s.message; // TS knows message exists!
  }
}

See Lesson 06 — Type Narrowing & Guards.

5Generics & Utility Types

Reusable abstractions
// Generic function
function wrapInArray<T>(item: T): T[] {
  return [item];
}

interface Product {
  id: number;
  name: string;
  price: number;
}

// Built-in utilities
type ReadonlyProduct = Readonly<Product>;
type ProductUpdate = Partial<Product>; // all fields optional
type ProductPreview = Pick<Product, "id" | "name">;
type WithoutPrice = Omit<Product, "price">;

See Lesson 08 — Utility Types & Generics.
