1Join Type Matrix

Visualizing set intersections
-- INNER JOIN: Intersection only (both sides must match)
SELECT * FROM users u INNER JOIN orders o ON u.id = o.user_id;

-- LEFT JOIN: All users + matching orders (or NULLs if no orders)
SELECT * FROM users u LEFT JOIN orders o ON u.id = o.user_id;

-- FULL OUTER JOIN: All users + all orders (matched where possible)
SELECT * FROM users u FULL OUTER JOIN orders o ON u.id = o.user_id;

See Lesson 02 — Inner joins versus outer joins.

2Common Table Expressions (CTEs)

Readable multi-step queries
WITH monthly_sales AS (
  SELECT user_id, SUM(amount) AS total_spent
  FROM orders
  WHERE order_date >= '2026-01-01'
  GROUP BY user_id
)
SELECT u.name, COALESCE(ms.total_spent, 0) AS revenue
FROM users u
LEFT JOIN monthly_sales ms ON u.id = ms.user_id
ORDER BY revenue DESC;

See Lesson 05 — Subqueries and Common Table Expressions (CTEs).

3Self-Joins & Hierarchies

Joining a table to itself
-- Find each employee and their direct manager
SELECT e.name AS employee, m.name AS manager
FROM employees e
LEFT JOIN employees m ON e.manager_id = m.id;

See Lesson 06 — Self-joins and hierarchical trees.

4Window Functions

Analytics without collapsing rows
SELECT id, user_id, amount,
  -- Running total per user
  SUM(amount) OVER (PARTITION BY user_id ORDER BY order_date) AS running_total,
  -- Order ranking per user
  ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY amount DESC) AS spending_rank
FROM orders;
