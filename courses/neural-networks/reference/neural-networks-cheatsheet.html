2Residual Skip Connection Pattern

Uninterrupted gradient highway
import torch.nn as nn
class ResBlock(nn.Module):
    def __init__(self, layer): super().__init__(); self.layer = layer
    def forward(self, x): return self.layer(x) + x  # The '+ x' is the highway!

4Softmax Multi-Class Probability

Logits to probabilities
import numpy as np
def softmax(logits):
    exp_s = np.exp(logits - np.max(logits, axis=-1, keepdims=True))
    return exp_s / np.sum(exp_s, axis=-1, keepdims=True)
