1Minimal FastMCP Server (Python)

Instant local tool server
from mcp.server.fastmcp import FastMCP

mcp = FastMCP("MyTools")

@mcp.tool()
def fetch_weather(city: str) -> str:
    """Get current weather conditions for a city."""
    return f"Weather in {city}: 22C, Sunny"

if __name__ == "__main__":
    mcp.run(transport="stdio")

4Filesystem Root Scoping Check

Path traversal defense
def check_root_security(requested_path, allowed_root):
    real_path = os.path.realpath(requested_path)
    if not real_path.startswith(os.path.realpath(allowed_root)):
        raise PermissionError('Access denied: Outside allowed root!')
