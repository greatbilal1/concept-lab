1Minimal FastAPI LLM Endpoint

Production starter pattern
from fastapi import FastAPI
from pydantic import BaseModel
from openai import OpenAI

app = FastAPI()
client = OpenAI()

class Query(BaseModel):
    prompt: str

@app.post("/api/generate")
def generate(q: Query):
    res = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": q.prompt}],
        temperature=0.3
    )
    return {"result": res.choices[0].message.content}

2Resilient Retry Decorator (Tenacity)

Exponential backoff with jitter
from tenacity import retry, stop_after_attempt, wait_exponential
import openai

@retry(stop=stop_after_attempt(3), wait=wait_exponential(multiplier=1, min=2, max=10))
def call_with_retry(prompt):
    return client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": prompt}],
        timeout=15.0
    )

3Real-Time Streaming Generator

Iterating over chunks
stream = client.chat.completions.create(
    model="gpt-4o-mini",
    messages=[{"role": "user", "content": "Explain RAG"}],
    stream=True
)
for chunk in stream:
    content = chunk.choices[0].delta.content
    if content: print(content, end="", flush=True)

4Defensive JSON Markdown Stripper

Cleaning code fences before parsing
import re, json
def parse_clean_json(text: str) -> dict:
    # Strip ```json ... ``` code fences:
    cleaned = re.sub(r"^```(?:json)?\n?|\n?```$", "", text.strip())
    return json.loads(cleaned)
